function Themechanger() {
    const [cname,setCname] = useState('light');
    const [text,setText] = useState("Dark")
    function handleClick(){

            setCname('dark');

import React, { useState } from 'react'

export default function App() {
  const [username,setUsername] = useState("")
  const [pass,setPass] = useState("")
  const [condition,setConditon ] = useState(false)
  function handleForm(e){
    e.preventDefault()
    setConditon(true)
  }
  return (
    <div>
      <form onSubmit={handleForm}>
        <input type="text"
         placeholder='Enter username' 
         value={username}
         onChange={(event)=>setUsername(event.target.value)}/>
        <input type="pass" 
        placeholder='Enter Password' 
        value={pass}
        onChange={(e)=>setPass(e.target.value)}/>
        <button>Login</button>
        {condition && (
          <>
           <h1>I am {username} and my password is {pass}</h1>
          </>
        )}
      </form>
    </div>
  )
}


import React, { useState } from 'react'

export default function App() {
  const [data,setData] = useState({
    username:"",
    email:"",
    pass:""
  })
  const [condition,setConditon ] = useState(false)
  function handleForm(e){
    e.preventDefault()
    setConditon(true)
  }
  return (
    <div>
      <form onSubmit={handleForm}>
        <input type="text"
         placeholder='Enter username' 
         value={data.username}
         onChange={(event)=>{setData((prev)=>({...prev,username:event.target.value}))}}/>
         <input type="email" placeholder='Enter email'
         value={data.email}
         onChange={(e)=>{setData((prev)=>({...prev,email:e.target.value}))}}/>
        <input type="pass" 
        placeholder='Enter Password' 
        value={data.pass}
        onChange={(e)=>{setData((prev)=>({...prev,pass:e.target.value}))}}/>
        <button>Login</button>
        {condition && (
          <>
           <h1>I {data.email} am {data.username} and my password is {data.pass}</h1>
          </>
        )}
      </form>
    </div>
  )
}
